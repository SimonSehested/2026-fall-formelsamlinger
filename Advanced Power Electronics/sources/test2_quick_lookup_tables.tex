\textbf{Two-Transistor Forward} &
$M = \dfrac{N_2}{N_1} D$ &
$V_{sw} = V_{in}$ &
$V_{D1} = \dfrac{N_2}{N_1} V_{in}$ \newline
$V_{D2} = \dfrac{N_2}{N_1} V_{in}$ &
$D < 0.5$. Magnetizing energy clamped to $V_{in}$ via 2 primary diodes. Switches see only $V_{in}$; ideal for high $V_{in}$ bus. \\

\textbf{Forward (Overlap)} &
$V_o = \dfrac{N_2}{N_1} D V_{in} - \Delta V_o$ \newline
$\Delta V_o = f_s L_k I_o = \left(\dfrac{N_2}{N_1} V_{in}\right)\dfrac{t_{ov}}{T_s}$ &
$V_{sw} = V_{in}\left(1 + \dfrac{N_1}{N_3}\right)$ &
$V_{D1} = \dfrac{N_2}{N_1} V_{in}$ \newline
$V_{D2} = \dfrac{N_2}{N_1} V_{in}\left(1 + \dfrac{N_1}{N_3}\right)$ &
Leakage $L_k$ delays transfer of $I_o$, causing both diodes to conduct during $t_{ov} = \dfrac{I_o L_k}{V_{in}(N_2/N_1)}$. Duty loss $\Delta D = \dfrac{t_{ov}}{T_s}$. \\

\textbf{Phase-Shifted Full-Bridge} &
$V_o = \dfrac{N_2}{N_1} D_{eff} V_{in}$ \newline
$D_{eff} = D - \Delta D$ &
$V_{sw} = V_{in}$ &
$V_D = 2 \left(\dfrac{N_2}{N_1}\right) V_{in}$ \newline
(center-tap secondary) &
$\Delta D = \dfrac{4 f_s L_{lk} I_o}{n V_{in}}$ with $n = N_2/N_1$. Primary switches achieve ZVS via $L_{lk}$ and $C_{oss}$. Constant $50\%$ duty. \\

\textbf{Dual-Active-Bridge (DAB)} &
$P = \dfrac{V_1 V_2}{2\pi f_s L_k}\phi\left(1 - \dfrac{|\phi|}{\pi}\right)$ \newline
$= \dfrac{V_1 V_2}{2 f_s L_k} D_\phi (1 - |D_\phi|)$ \newline
for $\phi \in [-\pi/2, \pi/2]$ &
$V_{sw1} = V_1$ \newline
$V_{sw2} = V_2$ &
--- (Active H-bridges) &
$P_{\max} = \dfrac{\pi V_1 V_2}{4 \omega L_k} = \dfrac{V_1 V_2}{8 f_s L_k}$ at $\phi = \pi/2$ ($D_\phi = 0.50$). Bidirectional power transfer. Inherent soft-switching (ZVS). \\

\noindent\textbf{Power Quality Metrics \& Boost PFC Sizing Summary:} \par\vspace{1pt}
\noindent$\bullet$ \textbf{Power Factor Decomposition:} $PF = \dfrac{P}{S} = \left(\dfrac{I_{s1,rms}}{I_{s,rms}}\right) \cos\phi_1 = DF \times DPF = \dfrac{\cos\phi_1}{\sqrt{1 + THD^2}}$, where $DF = \dfrac{1}{\sqrt{1 + THD^2}}$, $DPF = \cos\phi_1$. \par\vspace{1pt}
\noindent$\bullet$ \textbf{THD:} $THD = \dfrac{\sqrt{I_{s,rms}^2 - I_{s1,rms}^2}}{I_{s1,rms}} = \sqrt{\dfrac{1}{DF^2} - 1}$. \par\vspace{1pt}
\noindent$\bullet$ \textbf{Boost PFC Inductor Sizing Rule:} $L \ge \dfrac{\sqrt{2}V_{in}(V_o - \sqrt{2}V_{in})}{f_s \Delta I_L V_o} = \dfrac{\eta V_{in}^2 (V_o - \sqrt{2}V_{in})}{f_s \cdot r \cdot P_o V_o}$ ($V_{in}$ line RMS, $\sqrt{2}V_{in}$ peak input, $V_o > \sqrt{2}V_{in}$, ripple $\Delta I_L = r \hat{I}_{in}$, load $P_o$, efficiency $\eta$).

\noindent\textbf{Output Low-Pass Filter Design Guideline:}
$10 f_1 < f_c = \dfrac{1}{2\pi\sqrt{L_f C_f}} < \dfrac{1}{10} f_s$.
Ensures negligible attenuation/phase lag at fundamental grid/load frequency $f_1$ (\SI{50}{\hertz}/\SI{60}{\hertz}), while suppressing carrier harmonics at $f_s$ by $\ge \SI{40}{\decibel\per decade}$.

\noindent\textbf{State-Space Averaging (SSA) Definitions ($d = D + \hat{d}$, $x = X + \hat{x}$):}
$A = D A_1 + (1-D) A_2$, $B = D B_1 + (1-D) B_2$, $F = (A_1 - A_2)X + (B_1 - B_2)V_{in}$,
$\hat{x}(s) = (sI - A)^{-1} F \hat{d}(s)$. \par\vspace{1pt}
\noindent\textbf{Canonical Form:} $G_{vd}(s) = \dfrac{G_{vd0} \left(1 - \frac{s}{\omega_z}\right)}{1 + \frac{s}{Q \omega_0} + \left(\frac{s}{\omega_0}\right)^2} = \dfrac{G_{vd0} \left(1 - \frac{s}{\omega_z}\right)}{1 + \frac{2\zeta}{\omega_0} s + \frac{s^2}{\omega_0^2}}$.

$B_1 = \begin{bmatrix} 1/L \\ 0 \end{bmatrix}, \; B_2 = \begin{bmatrix} 0 \\ 0 \end{bmatrix}$ \newline
